================================================================
  8. ENDPOINT API (dari Backend Pterodactyl)
================================================================

BASE URL: http://<SERVER_HOST>:<PORT>/api

ENDPOINT WAJIB:
  GET /api/healthz
    Response: { status: "ok", uptime: "6m" }

  GET /api/signals
    Response: Array<Signal>
    Query params: ?pair=XAUUSD &status=OPEN &limit=20

  GET /api/signals/latest
    Response: Signal (sinyal paling baru)

  GET /api/stats
    Response: {
      totalSignals: number,
      winRate: number,
      wins: number,
      losses: number,
      totalPips: number,
      avgPips: number,
      lastSignal: string,
      closed: number,
      open: number
    }

  GET /api/performance
    Response: Array<{ date: string, pips: number }>  // 7–30 hari
    Query params: ?range=7d &range=30d &range=3m

  GET /api/performance/best
    Response: Array<{ pair: string, pips: number }>

  GET /api/server/resources
    Response: { cpu: string, memory: string, disk: string, uptime: string, status: "running"|"stopped" }

  GET /api/journal
    Response: Array<JournalEntry>

ENDPOINT OPSIONAL (halaman lanjutan):
  GET /api/backtest              → Array<BacktestResult>
  GET /api/checklist/latest      → { h4trend, h1trend, confidence, status }
  GET /api/reports/daily         → DailyReport
  GET /api/reports/weekly        → WeeklyReport

RESPONSE SHAPE — Signal (JSON):
  {
    "id": 1,

  14. CONTOH SNIPPET KONFIGURASI API (TypeScript)
================================================================

// src/config/api.ts
export const API_BASE =
  process.env.NEXT_PUBLIC_API_URL ||
  'http://ndserverku.lynzzofficial.com:2389';

// src/hooks/useSignals.ts
import { useQuery } from '@tanstack/react-query';
import { API_BASE } from '../config/api';

export function useSignals(pair?: string) {
  return useQuery({
    queryKey: ['signals', pair],
    queryFn: async () => {
      const url = pair
        ? `${API_BASE}/api/signals?pair=${pair}`
        : `${API_BASE}/api/signals`;
      const res = await fetch(url);
      if (!res.ok) throw new Error('Failed to fetch signals');
      return res.json();
    },
    refetchInterval: 30_000,
    staleTime: 10_000,
  });
}

// src/hooks/useStats.ts
export function useStats() {
  return useQuery({
    queryKey: ['stats'],
    queryFn: async () => {
      const res = await fetch(`${API_BASE}/api/stats`);
      if (!res.ok) throw new Error('Failed to fetch stats');
      return res.json();
    },
    refetchInterval: 30_000,
  });
}
